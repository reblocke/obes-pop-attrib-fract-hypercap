Below is my first (probably most stringent) iteration of hypercapnic respiratory failure: co2 greater than 45 if abg, 50 if vbg, ph <7.35 - they have to have either an abg or a vbg showing this, AND either NIV or IMV or both. I’ve written it so that ABG values and VBG values cannot mix

stata_data <- stata_data %>%
  mutate(
    gas_source = case_when(
      !is.na(highest_paco2) & !is.na(abg_ph) ~ "ABG",
      TRUE ~ "VBG"
    )
  )
stata_data <- stata_data %>%
  mutate(
    hrf_hypercapnic = case_when(
      # ABG-based HRF
      (niv_proc == 1 | imv_proc == 1) &
        gas_source == "ABG" &
        highest_paco2 >= 45 &
        abg_ph <= 7.35 ~ 1,

      # VBG-based HRF (only when ABG incomplete)
      (niv_proc == 1 | imv_proc == 1) &
        gas_source == "VBG" &
        !is.na(vbg_co2) & !is.na(vbg_ph) &
        vbg_co2 >= 50 &
        vbg_ph <= 7.35 ~ 1,

      TRUE ~ 0
    ),
    hrf_hypercapnic = factor(hrf_hypercapnic, labels = c("No", "Yes"))
  )

Regression done with new definition of hypercapnic respiratory failure - pH and CO2 definition as above, except now including (required) ICD codes instead of ventilation (I only used 4 - the first four that are listed in database):

stata_data <- stata_data %>%
  mutate(
    # Lab-based hypercapnic respiratory failure (ABG or VBG)
    hrf_hypercapnic_labs_v2 = case_when(
      !is.na(highest_paco2) & !is.na(abg_ph) &
       highest_paco2 >= 50 & abg_ph < 7.35 ~ 1,
      
      !is.na(vbg_co2) & !is.na(vbg_ph) &
        vbg_co2 >= 50 & vbg_ph < 7.35 ~ 1,
      
      TRUE ~ 0
    ),
    
    # ICD-based hypercapnic respiratory failure
    hrf_hypercapnic_icd_v2 = ifelse(
      has_j9612 == 1 |
      has_j9622 == 1 |
      has_j9692 == 1 |
      has_j9602 == 1,
      1, 0
    ),
    
    # Final combined HRF definition (labs OR ICD)
    hrf_hypercapnic_v2 = ifelse(
      hrf_hypercapnic_labs_v2 == 1 |
      hrf_hypercapnic_icd_v2 == 1,
      1, 0
    )
  )
stata_data <- stata_data %>%
  mutate(
    hrf_hypercapnic_v2_strict = ifelse(
      hrf_hypercapnic_labs_v2 == 1 &
      hrf_hypercapnic_icd_v2  == 1,
      1, 0
    )
  )

  filter(
